```python
class ListNode:
        def __init__(self, val=None, next=None):
            self.val = val
            self.next = next

class Solution:
    def isValid(self, s: str) -> bool:
        close_to_open = {
            ")": "(",
            "}": "{",
            "]": "[",
        }
        i = 0
        dummy = ListNode(None)
        bracket_log = dummy
        while i < len(s):
            j = 0
            # creat a open bracket log
            while i+j < len(s):
                bracket = s[i+j]
                if bracket in close_to_open:
                    break
                bracket_log = ListNode(bracket, bracket_log)
                j += 1
            if i+j == len(s) or bracket_log.val is None:
                return False
            # validate correspondence 
            while bracket_log.val is not None:
                if i+j == len(s) and bracket_log.val != None:
                    return False
                bracket = s[i+j]
                if bracket not in close_to_open:
                    break
                if bracket_log.val != close_to_open[bracket]:
                    return False
                bracket_log = bracket_log.next
                j += 1
            i += j
        return True
```

あれ？　ListNodeである必要なくないか？ List.popでうまくいく気がする。
```python
class Solution:
    def isValid(self, s: str) -> bool:
        close_to_open = {
            ")": "(",
            "}": "{",
            "]": "[",
        }
        open_brackets = []
        i = 0
        while i < len(s):
            bracket = s[i]
            if bracket not in close_to_open:
                open_brackets.append(bracket)
            else:
                if not open_brackets or close_to_open[bracket] != open_brackets[-1]:
                    return False
                open_brackets.pop(-1)
            i += 1
        if open_brackets:
            return False
        return True
```

先に書いたコードと比べると、close_to_openをopen_to_closeにした。
その方が何だか決まりがいい。
```python
class Solution:
    def isValid(self, s: str) -> bool:
        open_to_close = {
            "(": ")",
            "{": "}",
            "[": "]",
            "*": ""
        }
        stack = ["*"]
        for c in s:
            if c in open_to_close:
                stack.append(c)
            if c in open_to_close.values():
                if c != open_to_close[stack[-1]]:
                    return False
                else:
                    stack.pop()
        return stack == ["*"]
```
