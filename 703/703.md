```python
class KthLargest:

    def __init__(self, k: int, nums: List[int]):
        self.k = k
        self.stream = sorted(nums) 
        self.top_k = self.initialize_top_k()
        self.kth_largest = self.top_k[-1]


    def initialize_top_k(self):
        if len(self.stream) >= self.k:
            top_k = [self.stream[-1 * i] for i in range(1, self.k + 1)]
            return top_k
        else:
            top_k = []
            top_k.extend(self.stream)
            shortage = self.k - len(self.stream)
            dummy = [-1 * float('inf') for i in range(shortage)]
            top_k.extend(dummy)
            return sorted(top_k, reverse=True)


    def add(self, val: int) -> int:
        if self.kth_largest > val:
            self.stream.append(val)
            return self.kth_largest
        else:
            self.top_k.pop()
            self.top_k.append(val)
            self.top_k.sort(reverse=True)
            self.kth_largest = self.top_k[-1]
        return  self.kth_largest
```
