```python
class Solution:
    def numUniqueEmails(self, emails: List[str]) -> int:
        canonical_addresses = set()
        for addr in emails:
            local, domain = addr.rsplit('@', maxsplit=1)
            # identify canonical
            t = []
            for c in local:
                if c == '.':
                    continue
                if c == '+':
                    break
                t.append(c)
            canon = ''.join(t) + '@' + domain
            # enroll
            canonical_addresses.add(canon)
        
        return len(canonical_addresses)
```

### Step 3
```python
class Solution:
    def is_valid_letters(
        self,
        letters: str, 
        special_cases: List[str] = ['+', '.', '@']
    ) -> bool:
        return all(
            ord('a') <= ord(l) <= ord('z') or l in special_cases 
            for l in letters
        )

    def is_valid_address(self, local: str, domain: str) -> bool:
        # chack a local part
        if not self.is_valid_letters(local):
            return False
        if local.startswith('+'):
            return False
        # check a domain part
        if not self.is_valid_letters(domain, ['.', '+']):
            return False
        before, after = domain.rsplit('.com', maxsplit=1)
        if not before:
            return False
        if after:
            return False

        return True

    def numUniqueEmails(self, emails: List[str]) -> int:
        unique_emails = set()
        for addr in emails:
            local, domain = addr.rsplit('@', maxsplit=1)
            # Except invalid addresses
            if not self.is_valid_address(local, domain):
                continue
            # Identify a canonical form
            t = []
            for c in local:
                if c == '.':
                    continue
                if c == '+':
                    break
                t.append(c)
            canon = ''.join(
                [''.join(t), '@', domain]
            )
            # Enroll
            unique_emails.add(canon)

        return len(unique_emails)
```
