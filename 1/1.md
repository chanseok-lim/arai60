```python
class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        pairs = {}
        for n in nums:
            m = target - n
            pairs[str(n)] = m
        
        for k, v in pairs.items():
            if v in nums:
                return [nums.index(int(k)), nums.index(v)]
```

```
        hashmap = {}
        for i in range(len(nums)):
            complement = target - nums[i]
```
