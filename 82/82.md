おそらく、扱うnodeが２つあるのと、本体とそのnext（イメージでは腕）を別々に操作しないといけないのとで難しい。
理解できたので書けた。
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        dummy = ListNode(1j, head) # edge case handling: duplicates at the head
        node = dummy
        forward = node.next
        while forward is not None:
            if forward.next is None or forward.val != forward.next.val:
                forward = forward.next
                node = node.next
                continue
            duplicate_number = forward.val
            while forward is not None and forward.val == duplicate_number:
                forward = forward.next
                node.next = forward
        return dummy.next
```
時間計算量: O(n), n = length of list<br>
実行時間: 300 / 10^6 = 3 * 10^{-4}<br>
空間計算量: O(1)<br>

```python
# cited from https://github.com/nittoco/leetcode/pull/9/files#r1634077440
class Solution:
    def deleteDuplicates(self, head: Optional[ListNode]) -> Optional[ListNode]:
        def iter_list(node):
            while node:
                yield node
                node = node.next

        sentinel = ListNode()
        tail = sentinel
        for k, g in groupby(iter_list(head), key=lambda node:node.val):
            first = next(g)
            if next(g, None):
                continue
            tail.next = first
            tail = tail.next
        tail.next = None
        return sentinel.next
```
