## Step 1
思いついたままに書く。前回頭を悩ませたおかげか、今回はdummyを使えばいいとすぐにわかった。
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def set_in_ascending_order(
        self, l1: Optional[ListNode], l2: Optional[ListNode]) \
        -> (Optional[ListNode], Optional[ListNode]
    ):
        node1, node2 = l1, l2
        while node1 is not None and node2 is not None:
            node1 = node1.next
            node2 = node2.next
            if node2 is None:
                return (l2, l1)
        return (l1, l2)

    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        l1, l2 = self.set_in_ascending_order(l1, l2)
        dummy = ListNode(-1, None)
        node = dummy
        carry = 0
        while l1 is not None:
            val = (l1.val + l2.val + carry) % 10
            carry = (l1.val + l2.val + carry) // 10
            node.next = ListNode(val, None)
            l1 = l1.next
            l2 = l2.next
            node = node.next
        while l2 is not None:
            val = (l2.val + carry) % 10
            carry = (l2.val + carry) // 10
            node.next = ListNode(val, None)
            l2 = l2.next
            node = node.next
        if carry != 0:
            node.next = ListNode(carry, None)
        return dummy.next
```
時間計算量: O(n), n = max(len(l1), len(l2))<br>
実行時間: 10^2 / 10^6 = 10^{-4} s<br>
空間計算量: O(n)<br>

```python
    def linked_list_to_integer(self, head: Optional[ListNode]) -> int:
        total = 0
        digit = 1
        while head is not None:
            total += head.val * digit
            head = head.next
            digit *= 10
        return total
    
    def integer_to_linked_list(self, n: int) -> Optional[ListNode]:
        if n == 0:
            return ListNode(0, None)
        dummy = ListNode(-1, None)
        node = dummy
        digit = 10
        quotient = n
        while quotient > 0:
            remainder = quotient % digit
            quotient = quotient // digit
            node.next = ListNode(remainder, None)
            node = node.next
        return dummy.next

        n1 = self.linked_list_to_integer(l1)
        n2 = self.linked_list_to_integer(l2)
```

```python
# cited from https://github.com/hayashi-ay/leetcode/blob/hayashi-ay-patch-13/2.%20Add%20Two%20Numbers.md
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        sentinel = ListNode()
        current = sentinel
        carry = 0
        while l1 or l2 or carry:
            val1 = l1.val if l1 else 0
            val2 = l2.val if l2 else 0
            total = val1 + val2 + carry
            current.next = ListNode(total % 10)
            carry = total // 10
            current = current.next
            l1 = l1.next if l1 else None
            l2 = l2.next if l2 else None
        return sentinel.next
```

## Step 3
```python
# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        def get_val(node: Optional[ListNode]) -> int:
            if node is None:
                return 0
            return node.val

        dummy = ListNode(None)
        node = dummy
        carry = 0
        while l1 is not None or l2 is not None or carry != 0:
            v1 = get_val(l1)
            v2 = get_val(l2)
            total = v1 + v2 + carry
            node.next = ListNode(total % 10)
            carry = total // 10
            if l1 is not None:
                l1 = l1.next
            if l2 is not None:
                l2 = l2.next
            node = node.next
        return dummy.next
```
最後の最後まで`node = node.next`と`carry = total // 10`の位置に悩む。<br>
とりあえず「次のステップへ向けての処理」でまとめてこの位置にする。
