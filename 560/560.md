class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        subarrays = []
        for i in range(len(nums)):
            subarr = []

            for n in nums[i:]:
                subarr.append(n)

                if sum(subarr) == k:
                    subarrays.append(subarr)
                elif sum(subarr) > k:
                    continue

        return len(subarrays)
