class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        subarrays = []
        for i in range(len(nums)):
            subarr = []

            for n in nums[i:]:
                subarr.append(n)

                if sum(subarr) == k:
                    subarrays.append(subarr)
                elif sum(subarr) > k:
                    continue

        return len(subarrays)
```

class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        assert isinstance(nums, list) and len(nums) != 0, f'expected: non-empty list object, actual: {nums}'
        prefix_sum_to_count = defaultdict(int)
        prefix_sum_to_count[0] = 1
        s_i = 0 # let s_i be the i-th prefix sum
        result = 0
        for n in nums:
            s_i += n
            result += prefix_sum_to_count[s_i - k]
            prefix_sum_to_count[s_i] += 1

        return result
```

## Step 3
コメントや変数名を整えた
```python
class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        if not isinstance(nums, list) or not nums:
            raise ValueError(f'Expected: non-empty list, Actual: {nums}')
        if not isinstance(k, int):
            raise ValueError(f'Expected: int, Actual: {k}')

        prefix_sum_to_count = defaultdict(int)
        prefix_sum_to_count[0] = 1 # Handle cases like nums = [0, ...], k = 0
        cur_sum = 0 # the prefix sum up to the current element
        subarr_count = 0
        for n in nums:
            cur_sum += n
            subarr_count += prefix_sum_to_count[cur_sum - k]
            prefix_sum_to_count[cur_sum] += 1

        return subarr_count
```
時間計算量: O(n), n = len(nums)<br>
実行時間: 2 * 10^4 / 10^6 = 2 * 10^{-1}<br>
空間計算量: O(n)
