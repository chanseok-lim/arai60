```python
class Solution:
    def filter_first_unique(
        self, char_to_indeces: Dict[str, List[int]]
    ) -> (str, int):
        unique_chars = []
        for c, inds in char_to_indeces.items():
            if len(inds) == 1:
                unique_chars.append((c, *inds))
        unique_chars = sorted(unique_chars, key=lambda x: x[1])

        if not unique_chars:
            return (None, -1)
        else:
            return unique_chars[0]

    def firstUniqChar(self, s: str) -> int:
        char_to_indeces = defaultdict(list)
        for i, c in enumerate(s):
            char_to_indeces[c].append(i)
        _, i = self.filter_first_unique(char_to_indeces)
        
        return i
```

```python
class Solution:
    def firstUniqChar(self, s: str) -> int:
        # 再帰で
        def first_uniq_char(s, length):
            if not s:
                unique_stack = []
                char_to_index = {}
                return char_to_index, unique_stack

            char = s[0]
            char_to_index, unique_stack = first_uniq_char(s[1:], length)
            
            if char not in char_to_index:
                char_to_index[char] = length - len(s)
                unique_stack.append(length - len(s))
                return char_to_index, unique_stack
                
            if char_to_index[char] is None:
                return char_to_index, unique_stack

            unique_stack.remove(char_to_index[char])
            char_to_index[char] = None
            return char_to_index, unique_stack

        _, unique_stack = first_uniq_char(s, len(s))

        if not unique_stack:
            return -1
        return unique_stack.pop()
```

## Step 3
```python
class Solution:
    def in_valid_characters(self, s: str) -> bool:
        return all(
            ord('a') <= ord(c) <= ord('z') for c in s
        )

    def firstUniqChar(self, s: str) -> int:
        if not self.in_valid_characters(s):
            raise ValueError(
                f'{s} contains characters other than lowercase English letters'
            )

        char_to_counts = Counter(s)

        for i, c in enumerate(s):
            if char_to_counts[c] == 1:
                return i

        return -1 # not found
```
時間計算量：O(n), n = len(s)
実行時間：10^5 / 10^6 = 10^{-1}
空間計算量：O(1), 26文字の定数倍
